const fs = require("fs")
const DB_FILE = "./database.json"

function loadDB() {
return JSON.parse(fs.readFileSync(DB_FILE))
}

function saveDB(data) {
fs.writeFileSync(DB_FILE, JSON.stringify(data, null, 2))
}

function getUser(id) {
const db = loadDB()
if (!db[id]) {
db[id] = {
money: 1000,
bank: 0,
inventory: [],
cars: [],
houses: []
}
saveDB(db)
}
return db[id]
}

module.exports = { getUser, loadDB, saveDB }
